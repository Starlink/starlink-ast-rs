```rust
/// An owned, dynamically-dispatched Region.
pub type DynRegion = Box<dyn Region>;

/// Result of comparing two regions for overlap.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Overlap {
    /// No overlap — regions are completely separate.
    None,
    /// The first region is entirely inside the second.
    FirstInside,
    /// The second region is entirely inside the first.
    SecondInside,
    /// Partial overlap — some shared area but neither contains the other.
    Partial,
    /// The regions are identical in extent.
    Same,
}

/// Boolean combination operator for CmpRegion.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CmpOperator {
    /// Intersection — points must be inside both regions.
    And,
    /// Union — points must be inside at least one region.
    Or,
}

/// A region of a coordinate frame.
///
/// Regions implement both [`Frame`] (they describe a coordinate system)
/// and [`Mapping`] (they transform points via the inside/outside predicate).
pub trait Region: Frame + Mapping {
    /// Test whether a point (in current frame) is inside this region.
    fn point_in_region(&self, point: &[f64]) -> AstResult<bool>;

    /// Determine the overlap between this region and another.
    fn overlap(&self, other: &dyn Region) -> AstResult<Overlap>;

    /// Return a negated copy of this region.
    fn negated_copy(&self) -> DynRegion;

    /// Negate this region in place (swap inside/outside).
    fn negate(&mut self);

    /// Whether this region is currently negated.
    fn is_negated(&self) -> bool;

    /// Whether the boundary is part of the region.
    fn is_closed(&self) -> bool;

    /// Set whether the boundary is part of the region.
    fn set_closed(&mut self, closed: bool);

    /// Fill factor (0.0..=1.0) — fraction of bounding box that is inside.
    fn fill_factor(&self) -> f64;

    /// Set the fill factor.
    fn set_fill_factor(&mut self, f: f64);

    /// Target point count for mesh generation.
    fn mesh_size(&self) -> usize;

    /// Set the mesh size.
    fn set_mesh_size(&mut self, n: usize);

    /// Whether this region has finite extent.
    fn is_bounded(&self) -> bool;

    /// Bounding box in current frame coordinates.
    fn get_region_bounds(&self, lbnd: &mut [f64], ubnd: &mut [f64]) -> AstResult<()>;

    /// Enclosing disc (centre + radius) for quick overlap pre-filtering.
    fn get_region_disc(&self, centre: &mut [f64], radius: &mut f64) -> AstResult<()>;

    /// Mesh of points on boundary or interior grid.
    fn get_region_mesh(&self, surface: bool) -> AstResult<Vec<Vec<f64>>>;

    /// The defining points of this region (meaning is class-specific).
    fn get_region_points(&self) -> AstResult<Vec<Vec<f64>>>;

    /// The current frame of this region.
    fn get_region_frame(&self) -> DynFrame;

    /// The full internal FrameSet (base frame → current frame).
    fn get_region_frameset(&self) -> FrameSet;

    /// Set the uncertainty region.
    fn set_uncertainty(&mut self, unc: Option<DynRegion>);

    /// Get the uncertainty region.
    fn get_uncertainty(&self, current: bool) -> Option<DynRegion>;

    /// Map this region into a new coordinate frame.
    fn map_region(&self, map: &dyn Mapping, frame: &dyn Frame) -> AstResult<DynRegion>;
}
```

```rust
//! Shared Region infrastructure: RegionData struct and RegionGeometry trait.

use std::cell::RefCell;

use crate::error::AstResult;
use crate::frames::FrameSet;
use crate::mappings::{DynMapping, Mapping};

use super::{DynRegion, Region};

/// Shared state owned by every concrete Region.
///
/// Contains the internal FrameSet (base frame where defining points live,
/// current frame which is the public interface), the defining points
/// themselves, and region-wide attributes.
#[derive(Debug, Clone)]
pub struct RegionData {
    /// Internal FrameSet: frame 1 = base frame, frame 2 = current frame.
    pub(crate) frameset: FrameSet,
    /// Defining points in base-frame coordinates.
    /// Outer vec = one entry per point, inner vec = coordinate values.
    pub(crate) points: Vec<Vec<f64>>,
    /// Whether inside/outside are swapped.
    pub(crate) negated: bool,
    /// Whether boundary points are considered "inside".
    pub(crate) closed: bool,
    /// Target mesh point count.
    pub(crate) mesh_size: usize,
    /// Fill factor (fraction of bounding box actually inside region).
    pub(crate) fill_factor: f64,
    /// Whether defining points adapt when frame attributes change.
    pub(crate) adaptive: bool,
    /// Optional uncertainty region.
    pub(crate) uncertainty: Option<Box<dyn Region>>,
    /// Cached base-frame boundary mesh.
    pub(crate) cached_base_mesh: RefCell<Option<Vec<Vec<f64>>>>,
}

impl RegionData {
    /// Create new RegionData with default attributes.
    pub(crate) fn new(frameset: FrameSet, points: Vec<Vec<f64>>) -> Self {
        Self {
            frameset,
            points,
            negated: false,
            closed: true,
            mesh_size: 200,
            fill_factor: 1.0,
            adaptive: true,
            uncertainty: None,
            cached_base_mesh: RefCell::new(None),
        }
    }

    /// Invalidate all cached derived data.
    pub(crate) fn invalidate_cache(&self) {
        *self.cached_base_mesh.borrow_mut() = None;
    }

    /// Number of axes in the base frame.
    pub(crate) fn naxes(&self) -> usize {
        self.frameset.base().naxes()
    }
}

/// Per-class geometry functions that each concrete Region must implement.
///
/// These are the Rust equivalent of the C virtual function pointers
/// (RegPins, RegBaseBox, RegBaseMesh, RegCentre, GetBounded).
pub(crate) trait RegionGeometry {
    /// Access the shared RegionData.
    fn region_data(&self) -> &RegionData;

    /// Mutable access to the shared RegionData.
    fn region_data_mut(&mut self) -> &mut RegionData;

    /// Batch predicate: test which points (in base frame coords) are inside.
    ///
    /// Returns a Vec<bool> with one entry per point. Points are provided
    /// as `points[axis][point_index]` (axis-major layout matching PointSet).
    fn reg_pins(&self, points: &[&[f64]], unc: Option<&dyn Region>) -> AstResult<Vec<bool>>;

    /// Tight axis-aligned bounding box in base frame coordinates.
    fn reg_base_box(&self, lbnd: &mut [f64], ubnd: &mut [f64]) -> AstResult<()>;

    /// Generate boundary mesh in base frame coordinates.
    ///
    /// Returns points as `result[axis][point_index]` (axis-major).
    fn reg_base_mesh(&self) -> AstResult<Vec<Vec<f64>>>;

    /// Centre of the region in base frame, if meaningful.
    fn reg_centre(&self) -> Option<Vec<f64>>;

    /// Whether the region has finite extent (before considering negation).
    fn is_bounded_inner(&self) -> bool;

    /// Remap defining points after an integrity delta.
    ///
    /// `base_to_old_current` maps base→old current frame.
    /// `old_to_new` maps old current→new current frame.
    /// `new_current_to_base` maps new current→base frame.
    ///
    /// Each class decides how to remap (Circle remaps centre and recomputes
    /// radius; Polygon remaps all vertices; etc).
    fn remap_defining_points(
        &mut self,
        base_to_old_current: &dyn Mapping,
        old_to_new: &dyn Mapping,
        new_current_to_base: &dyn Mapping,
    ) -> AstResult<()>;
}

/// Apply the region-as-mapping transform: passthrough inside, BAD outside.
///
/// This is the shared implementation of `Mapping::transform` for all Regions.
/// It transforms input points into the base frame, calls `reg_pins` to get
/// the inside/outside classification, applies negation, and produces output
/// where "inside" points pass through unchanged and "outside" points become
/// `BAD_VALUE`.
pub(crate) fn region_transform(
    geom: &dyn RegionGeometry,
    points: &crate::coords::PointSet,
    _forward: bool,
) -> AstResult<crate::coords::PointSet> {
    use crate::constants::BAD_VALUE;
    use crate::coords::PointSet;
    use crate::frames::FrameRef;

    let data = geom.region_data();
    let naxes = data.naxes();
    let npoint = points.npoint();

    // Transform input points into base frame (inverse of base→current mapping)
    let mapping = data.frameset.mapping(FrameRef::Base, FrameRef::Current)?;
    let base_points = mapping.transform(points, false)?;

    // Build axis-major slices for reg_pins
    let axis_slices: Vec<&[f64]> = (0..naxes)
        .map(|ax| base_points.axis(ax))
        .collect();

    // Call per-class predicate
    let inside = geom.reg_pins(&axis_slices, data.uncertainty.as_deref())?;

    // Apply negation
    let negated = data.negated;

    // Build output: passthrough for inside, BAD for outside
    let mut out = PointSet::new(naxes, npoint);
    for ax in 0..naxes {
        let in_axis = points.axis(ax);
        let out_axis = out.axis_mut(ax);
        for i in 0..npoint {
            let is_in = inside[i] ^ negated;
            out_axis[i] = if is_in { in_axis[i] } else { BAD_VALUE };
        }
    }

    Ok(out)
}
```

```rust
//! The Circle region — a circular or spherical region within a Frame.

use std::cell::Cell;

use crate::error::AstResult;
use crate::frames::{DynFrame, Frame, FrameSet};

use super::data::{RegionData, RegionGeometry};
use super::{DynRegion, Region};

/// A circular region within a coordinate frame.
#[derive(Debug, Clone)]
pub struct Circle {
    pub(crate) data: RegionData,
    pub(crate) cached_radius: Cell<Option<f64>>,
}

/// Builder for [`Circle`].
pub struct CircleBuilder {
    frame: DynFrame,
    centre: Vec<f64>,
    radius: f64,
}

impl CircleBuilder {
    /// Create a Circle from a centre point and radius.
    pub fn new(frame: impl Frame + 'static, centre: Vec<f64>, radius: f64) -> Self {
        Self {
            frame: Box::new(frame) as DynFrame,
            centre,
            radius,
        }
    }

    /// Build the Circle.
    pub fn build(self) -> AstResult<Circle> {
        // Compute edge point: centre + radius along axis 1
        let naxes = self.frame.naxes();
        let edge = self.frame.offset(&self.centre, &self.centre, self.radius)?;

        // Create internal FrameSet (base = current = supplied frame)
        let fs = FrameSet::new(self.frame.snapshot());

        let points = vec![self.centre, edge];
        let data = RegionData::new(fs, points);

        Ok(Circle {
            data,
            cached_radius: Cell::new(Some(self.radius)),
        })
    }
}
```

```rust
/// Execute a frame mutation with region integrity enforcement.
///
/// 1. Snapshot current frame
/// 2. Apply mutation via FrameSet integrity
/// 3. Compute delta mapping (old→new current)
/// 4. Remap defining points through the delta
/// 5. Invalidate caches
pub(crate) fn with_region_integrity<F>(
    geom: &mut dyn RegionGeometry,
    f: F,
) -> AstResult<()>
where
    F: FnOnce(&mut dyn Frame) -> AstResult<()>,
{
    let data = geom.region_data();
    let adaptive = data.adaptive;

    if !adaptive {
        // Non-adaptive: just mutate via FrameSet integrity (updates mapping only)
        geom.region_data_mut().frameset.with_integrity(f)?;
        return Ok(());
    }

    // Get base→old-current mapping before mutation
    let base_to_old = geom.region_data()
        .frameset
        .mapping(FrameRef::Base, FrameRef::Current)?;

    // Snapshot old current frame
    let snap = geom.region_data().frameset.current().snapshot();

    // Mutate via FrameSet integrity
    geom.region_data_mut().frameset.with_integrity(f)?;

    // Compute old→new delta
    let delta = geom.region_data().frameset.current().map_from(snap.as_ref())?;

    // Get new-current→base mapping (inverse of new base→current)
    let new_to_base = geom.region_data()
        .frameset
        .mapping(FrameRef::Current, FrameRef::Base)?;

    // Remap defining points
    geom.remap_defining_points(&*base_to_old, &*delta, &*new_to_base)?;

    // Invalidate cache
    geom.region_data().invalidate_cache();

    Ok(())
}
```

In `circle.rs`, implement:
```rust
fn remap_defining_points(
    &mut self,
    base_to_old_current: &dyn Mapping,
    old_to_new: &dyn Mapping,
    new_current_to_base: &dyn Mapping,
) -> AstResult<()> {
    // Transform centre: base → old-current → new-current → base
    let naxes = self.data.naxes();
    let mut pts = PointSet::new(naxes, 1);
    for (ax, val) in self.data.points[0].iter().enumerate() {
        pts.axis_mut(ax)[0] = *val;
    }
    let in_old = base_to_old_current.transform(&pts, true)?;
    let in_new = old_to_new.transform(&in_old, true)?;
    let in_base = new_current_to_base.transform(&in_new, true)?;

    for ax in 0..naxes {
        self.data.points[0][ax] = in_base.axis(ax)[0];
    }

    // Edge point similarly
    for (ax, val) in self.data.points[1].iter().enumerate() {
        pts.axis_mut(ax)[0] = *val;
    }
    let in_old = base_to_old_current.transform(&pts, true)?;
    let in_new = old_to_new.transform(&in_old, true)?;
    let in_base = new_current_to_base.transform(&in_new, true)?;

    for ax in 0..naxes {
        self.data.points[1][ax] = in_base.axis(ax)[0];
    }

    // Invalidate cached radius
    self.cached_radius.set(None);

    Ok(())
}
```

```rust
//! Shared bounding-box logic for Regions.

use crate::coords::PointSet;
use crate::error::AstResult;
use crate::frames::FrameRef;
use crate::mappings::Mapping;

use super::data::RegionGeometry;

/// Compute bounding box in current frame by transforming base-frame bounds.
///
/// Gets the tight base-frame box from the per-class `reg_base_box`,
/// generates corner points, transforms them to current frame, and takes
/// the axis-aligned enclosure.
pub(crate) fn get_region_bounds_impl(
    geom: &dyn RegionGeometry,
    lbnd: &mut [f64],
    ubnd: &mut [f64],
) -> AstResult<()> {
    let data = geom.region_data();
    let naxes = data.naxes();

    // Get base-frame bounds
    let mut base_lbnd = vec![0.0; naxes];
    let mut base_ubnd = vec![0.0; naxes];
    geom.reg_base_box(&mut base_lbnd, &mut base_ubnd)?;

    // For a simple (identity mapping) case, just return base bounds
    let mapping = data.frameset.mapping(FrameRef::Base, FrameRef::Current)?;

    // Generate 2^naxes corner points of the base bounding box
    let ncorners = 1usize << naxes;
    let mut pts = PointSet::new(naxes, ncorners);
    for corner in 0..ncorners {
        for ax in 0..naxes {
            let use_upper = (corner >> ax) & 1 == 1;
            pts.axis_mut(ax)[corner] = if use_upper {
                base_ubnd[ax]
            } else {
                base_lbnd[ax]
            };
        }
    }

    // Transform corners to current frame
    let current_pts = mapping.transform(&pts, true)?;

    // Take enclosing box
    for ax in 0..naxes {
        let axis = current_pts.axis(ax);
        let mut lo = f64::MAX;
        let mut hi = f64::MIN;
        for &val in axis {
            if val < lo {
                lo = val;
            }
            if val > hi {
                hi = val;
            }
        }
        lbnd[ax] = lo;
        ubnd[ax] = hi;
    }

    Ok(())
}
```

```rust
//! Shared mesh generation logic for Regions.

use crate::coords::PointSet;
use crate::error::AstResult;
use crate::frames::FrameRef;
use crate::mappings::Mapping;

use super::data::RegionGeometry;

/// Get the boundary mesh in current-frame coordinates.
///
/// Uses the cached base-frame mesh (or generates it via `reg_base_mesh`),
/// then transforms to the current frame.
pub(crate) fn get_region_mesh_impl(
    geom: &dyn RegionGeometry,
    _surface: bool,
) -> AstResult<Vec<Vec<f64>>> {
    let data = geom.region_data();
    let naxes = data.naxes();

    // Check cache
    {
        let cache = data.cached_base_mesh.borrow();
        if let Some(ref mesh) = *cache {
            // Transform cached mesh to current frame
            return transform_mesh_to_current(data, mesh, naxes);
        }
    }

    // Generate base mesh
    let base_mesh = geom.reg_base_mesh()?;

    // Cache it
    *data.cached_base_mesh.borrow_mut() = Some(base_mesh.clone());

    // Transform to current frame
    transform_mesh_to_current(data, &base_mesh, naxes)
}

fn transform_mesh_to_current(
    data: &super::data::RegionData,
    base_mesh: &[Vec<f64>],
    naxes: usize,
) -> AstResult<Vec<Vec<f64>>> {
    if base_mesh.is_empty() || base_mesh[0].is_empty() {
        return Ok(vec![Vec::new(); naxes]);
    }

    let npts = base_mesh[0].len();
    let mapping = data.frameset.mapping(
        crate::frames::FrameRef::Base,
        crate::frames::FrameRef::Current,
    )?;

    // Build PointSet from base mesh
    let mut pts = PointSet::new(naxes, npts);
    for ax in 0..naxes {
        pts.axis_mut(ax).copy_from_slice(&base_mesh[ax]);
    }

    // Transform
    let current_pts = mapping.transform(&pts, true)?;

    // Convert back to Vec<Vec<f64>>
    let mut result = Vec::with_capacity(naxes);
    for ax in 0..naxes {
        result.push(current_pts.axis(ax).to_vec());
    }

    Ok(result)
}
```

```rust
//! Generic masking: apply a Region to an N-dimensional data array.

use crate::coords::PointSet;
use crate::error::AstResult;
use crate::mappings::Mapping;

use super::Region;

/// Apply a region as a mask to a data array.
///
/// Iterates over pixel positions in the grid defined by `lbnd..=ubnd`,
/// transforms each position through `mapping` into the region's frame,
/// tests against the region predicate, and sets masked pixels to `val`.
///
/// Returns the number of pixels that were masked.
pub fn mask<T: Copy>(
    region: &dyn Region,
    mapping: &dyn Mapping,
    inside: bool,
    lbnd: &[i64],
    ubnd: &[i64],
    data: &mut [T],
    val: T,
) -> AstResult<usize> {
    let ndim = lbnd.len();
    let dims: Vec<usize> = (0..ndim)
        .map(|i| (ubnd[i] - lbnd[i] + 1) as usize)
        .collect();
    let total = dims.iter().product::<usize>();

    let mut count = 0;

    // Process in row-batches (along first axis)
    let row_len = dims[0];
    let nrows = total / row_len;

    for row in 0..nrows {
        // Compute multi-dimensional index for start of this row
        let mut idx = vec![0usize; ndim];
        let mut rem = row;
        for d in (1..ndim).rev() {
            idx[d] = rem % dims[d];
            rem /= dims[d];
        }

        // Build point set for this row
        let mut pts = PointSet::new(ndim, row_len);
        for col in 0..row_len {
            pts.axis_mut(0)[col] = (lbnd[0] + col as i64) as f64;
            for d in 1..ndim {
                pts.axis_mut(d)[col] = (lbnd[d] + idx[d] as i64) as f64;
            }
        }

        // Transform grid coords to region frame
        let region_pts = mapping.transform(&pts, true)?;

        // Test each point against the region
        let data_offset = row * row_len;
        for col in 0..row_len {
            let point: Vec<f64> = (0..ndim)
                .map(|d| region_pts.axis(d)[col])
                .collect();
            let is_inside = region.point_in_region(&point)?;
            let should_mask = if inside { is_inside } else { !is_inside };
            if should_mask {
                data[data_offset + col] = val;
                count += 1;
            }
        }
    }

    Ok(count)
}
```

Note: This initial version calls `point_in_region` per point. A performance
optimization (batch predicate) can be added later. Study
`reference/starlink-ast/src/region.c` `MaskD` (~line 3400+) for the
batch approach.
